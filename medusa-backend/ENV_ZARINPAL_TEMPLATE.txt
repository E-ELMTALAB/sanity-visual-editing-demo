# Copy these variables to your .env file and fill in your values

# Get this from: https://www.zarinpal.com/panel → Settings → API & Webhooks

# Use true during development and testing

# Local development example:
ZARINPAL_CALLBACK_URL=http://localhost:3000/checkout/callback

# ZARINPAL_CALLBACK_URL=https://yourdomain.com/checkout/callback
